########################################################################
# Semantic search configuration
########################################################################
# Semantic search finds items by meaning: extracted text, OCR, audio
# transcripts, chat messages read as conversation sequences, images and
# video frames. It is OFF by default. Turn it on in IPEDConfig.txt:
#   enableSemanticText   - text modality
#   enableSemanticImages - visual modality, images
#   enableSemanticVideos - visual modality, videos (by the frames already
#                          extracted for the video thumbnail)
#
# A semantic result is a RANKING, not a set: it never says "no results"
# and low proximity is not absence. Keyword search remains the authority
# for identifiers and for claims that something does not occur.
#
# Everything this capability stores lives in <case>/iped/semantic/. The
# main index is never changed.

########################################################################
# Model source, per modality (text / visual)
########################################################################
# 'local' runs the model on this workstation, with no network access.
# 'remote' sends content to a service you declare below. There is NO
# default destination. A case records the MODEL it was prepared with;
# questions may use another source as long as the model fingerprint
# confirms it is the same model.

textModelSource = local
# Relative to the IPED installation folder. Shipped with IPED.

########################################################################
# What enters the semantic indexes
########################################################################
# Categories to prepare, separated by ';'. Empty means every category
# (text: every item with text; visual: every image and video).
textCategories =
visualCategories =

# Media types of chat messages (parsers of the tree and UFDR instant messages;
# attachments are prepared as documents). Messages of these types are not prepared
# alone: they are read as conversation sequences of the same chat.
messageMediaTypes = message/x-whatsapp-message; message/x-telegram-message; message/x-skype-message; message/x-discord-message; message/x-threema-message; message/x-jabber-message; application/x-ufed-instantmessage

# Chat databases IPED decodes into chats and messages. Their own text is a listing
# of the conversations, which reach semantic search through the messages above, so
# they are recorded as "covered by the messages" instead of prepared as passages.
messageContainerMediaTypes = application/x-whatsapp-db; application/x-whatsapp-db-f; application/x-whatsapp-chatstorage; application/x-whatsapp-chatstorage-f; application/x-threema-chatstorage; application/x-threema-chatstorage-f; application/x-telegram-db; application/x-telegram-db-ios; application/x-ios-sms-db

# Text beyond this many characters per item is not prepared; the excess is
# declared in the item's coverage ("covered up to the limit").
maxTextCharsPerItem = 200000

# Passages: window size and overlap, in characters. Windows end at a
# sentence boundary when there is one nearby.
passageChars = 1000
passageOverlapChars = 200

# Conversation sequences: messages per window, step between windows,
# maximum characters per window, and the gap (minutes) that breaks a window.
chatWindowMessages = 8
chatWindowStride = 4
chatWindowMaxChars = 1500
chatMaxGapMinutes = 720
# A message shorter than this ("ok", an emoji) is never a sequence on its own: when a
# gap isolates it, it joins the neighbouring window of the same chat.
chatShortMessageChars = 20

# Images whose larger side is below this many pixels (icons, emoji) are not
# prepared, and appear in coverage with that reason.
minImageSize = 48

########################################################################
# Search
########################################################################
# Depth shown by default, and the ceiling the examiner (UI) or the agent
# (MCP, by paging) can extend to. The ceiling is a depth limit, not a
# relevance threshold. There is no proximity floor, by design.
defaultDepth = 100
maxDepth = 1000

# Passages retrieved per result entry before consolidation.
oversampling = 4

# A scope (current result) with up to this many items is searched exactly,
# comparing every item. Above it the search is approximate (HNSW), which is
# reproducible but may leave out an item an exhaustive comparison would
# include. The method used is declared in every result.
exactSearchMaxItems = 50000

# Questions longer than this are truncated, and the truncation is declared.
maxQueryChars = 2000

# Minimum cosine between the probes computed now and those recorded in the
# case for the model to count as the same model.
fingerprintTolerance = 0.9995

# Default ordering mode for text: 'hybrid' (meaning + words, fused by rank
# position) or 'semantic' (meaning only). Chosen per question in the UI and MCP.
defaultTextMode = hybrid
# Rank fusion constant: score = 1/(rrfK + semantic position)
#                              + rrfLexicalWeight/(rrfK + word position).
rrfK = 60
# Weight of the word-based list, in (0, 1]. Below 1, words reorder what meaning
# already found plausible instead of competing with it one to one. Measured on the
# evaluation corpus: 1.0 lost 36.6 points on questions with no exact term, 0.5 lost
# 9.8, 0.4 lost 2.4, 0.3 lost none (validation-log.md).
rrfLexicalWeight = 0.3
# Size of the word-based list before fusion; 0 means maxDepth x oversampling.
lexicalCandidates = 0

# Maximum characters returned by the MCP tool iped_semantic_context.
contextMaxChars = 8000

# HNSW graph parameters of the semantic indexes (recorded in the case).
hnswMaxConn = 16
hnswBeamWidth = 100

########################################################################
# Processing cost
########################################################################
# Inference threads; 0 means half of the available processors.
inferenceThreads = 0
# Passages embedded per model call.
batchSize = 32
# Pending passages. When the queue is full, processing workers wait: this
# bounds the memory the capability uses.
queueCapacity = 1024
# Commit interval of the semantic indexes during processing, in minutes.
commitIntervalMinutes = 10
